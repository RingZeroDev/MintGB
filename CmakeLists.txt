cmake_minimum_required(VERSION 3.10)
project(MintGB)
set(CMAKE_CXX_STANDARD 20)

add_executable(MintGB)

target_sources(MintGB PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp)

include(CTest)
enable_testing()

add_subdirectory(src/cartridge)
add_subdirectory(src/hardware)
add_subdirectory(src/ppu)
add_subdirectory(src/mmu)
add_subdirectory(src/cpu)
add_subdirectory(src/disassembler)

find_package(SDL3 REQUIRED)
target_link_libraries(MintGB PRIVATE SDL3::SDL3)
target_compile_definitions(MintGB PRIVATE SDL_MAIN_USE_CALLBACKS)
target_link_options(MintGB PRIVATE -static)

add_library(imgui)
target_include_directories(imgui 
    PUBLIC 
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/imgui
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/imgui/backends
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/imgui_club/imgui_memory_editor/
)
target_sources(imgui
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/imgui/imgui.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/imgui/imgui_demo.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/imgui/imgui_draw.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/imgui/imgui_tables.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/imgui/imgui_widgets.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/imgui/backends/imgui_impl_sdl3.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/thirdparty/imgui/backends/imgui_impl_sdlrenderer3.cpp
)
target_link_libraries(imgui PUBLIC SDL3::SDL3)

target_link_libraries(MintGB PRIVATE CPU MMU Cartridge imgui Disassembler)
