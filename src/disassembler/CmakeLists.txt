cmake_minimum_required(VERSION 3.10)
project(Disassembler)
set(CMAKE_CXX_STANDARD 20)

add_library(Disassembler STATIC)

target_sources(Disassembler
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src/disassembler.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/src/unprefixed_table.cpp
        ${CMAKE_CURRENT_SOURCE_DIR}/src/cb_table.cpp
)

target_include_directories(Disassembler
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/include/disassembler
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/include
)

target_link_libraries(Disassembler MMU)

add_executable(MintDasm)

target_sources(MintDasm
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp
)

target_include_directories(MintDasm
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/include/disassembler
)

target_link_libraries(MintDasm Disassembler Cartridge MMU)
